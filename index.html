Tooltip Demo with Code Snippet

This page demonstrates a custom tooltip functionality. When you hover over any element with a title attribute, the browser's native tooltip is disabled and replaced with a stylish, dynamic tooltip.

Copy & Paste Code Snippet

To use this tooltip functionality on your webpage, simply copy and paste the code snippet below into your HTML file. 

<style>
  .title-tooltip {
	position: relative;
	cursor: help;
  }
  .title-tooltip::after {
	content: var(--tooltip);
	position: fixed;
	left: calc(var(--tooltip-x, 0px) + 10px);
	top: calc(var(--tooltip-y, 0px) + 10px);
	background: rgba(51, 51, 51, 0.9);
	color: white;
	padding: 8px 10px;
	border-radius: 3px;
	white-space: normal;
	max-width: 200px;
	overflow-wrap: break-word;
	z-index: 100;
	opacity: 0;
	visibility: hidden;
	transform: translateY(10px);
	transition: opacity 0.3s ease, transform 0.3s ease;
  }
  .title-tooltip:hover::after {
	opacity: 1;
	visibility: visible;
	transform: translateY(0);
  }
</style>
<script>
  function measureTooltipHeight(text, refEl) {
	var temp = document.createElement("span");
	temp.style.position = "absolute";
	temp.style.visibility = "hidden";
	temp.style.padding = "8px 10px";
	temp.style.borderRadius = "3px";
	temp.style.whiteSpace = "normal";
	temp.style.maxWidth = "200px";
	temp.style.overflowWrap = "break-word";
	var computed = window.getComputedStyle(refEl);
	temp.style.font = computed.font;
	temp.style.background = "rgba(51, 51, 51, 0.9)";
	temp.style.color = "white";
	temp.textContent = text;
	document.body.appendChild(temp);
	var height = temp.offsetHeight;
	document.body.removeChild(temp);
	return height;
  }
  
  function initializeTitleTooltips() {
	document.querySelectorAll("[title]").forEach(function(el) {
	  var tooltipText = el.getAttribute("title");
	  el.style.setProperty("--tooltip", '"' + tooltipText + '"');
	  el.classList.add("title-tooltip");
	  el.removeAttribute("title");
	  
	  el.addEventListener("mousemove", function(e) {
		var offset = 10;
		var x = e.clientX + offset;
		var y = e.clientY + offset;
		if (!el._tooltipHeight) {
		  el._tooltipHeight = measureTooltipHeight(tooltipText, el);
		}
		var tooltipHeight = el._tooltipHeight;
		if (y + tooltipHeight + offset > window.innerHeight) {
		  y = y - tooltipHeight - offset - 20;
		  x = x - 20;
		}
		el.style.setProperty("--tooltip-x", x + "px");
		el.style.setProperty("--tooltip-y", y + "px");
	  });
	});
  }
  
  document.addEventListener("DOMContentLoaded", initializeTitleTooltips);
</script>

Simply include this code in your HTML file. Any element with a title attribute will now display the custom tooltip.
